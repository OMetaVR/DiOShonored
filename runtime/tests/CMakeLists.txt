# ctest suite. Every test is its own executable (own process: the guest address space is a process-wide singleton).
add_library(dish_test_support STATIC testing.cpp)
target_include_directories(dish_test_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${REPO_ROOT}/runtime/include ${REPO_ROOT}/include)

function(dish_test name)
  if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${ARGV1})
    return()
  endif()
  add_executable(${name} ${ARGN} $<TARGET_OBJECTS:dish_runtime>)
  target_link_libraries(${name} PRIVATE dish_test_support ${DISH_RUNTIME_LIBS})
  target_include_directories(${name} PRIVATE $<TARGET_PROPERTY:dish_runtime,INTERFACE_INCLUDE_DIRECTORIES>)   # SDL3 outside /usr/include
  target_link_options(${name} PRIVATE -rdynamic)
  target_compile_options(${name} PRIVATE -Wall -Wno-unused-parameter)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
endfunction()
